#Function to clean blanks in long format
clean_blanks_long <- function(df) {
  df %>%
    mutate(
      blank_type = case_when(
        str_detect(filename, "matrix_blank") ~ "matrix",
        str_detect(filename, "blank_[1-3]$") ~ "calibration",  #first 3 negative blanks
        str_detect(filename, "blank") ~ "solvent_blank",
        str_detect(filename, "pl_neg_1[7-9]$|pl_neg_20$|pl_neg_38$|pl_neg_40$") ~ "pl_control_blank",
        TRUE ~ "other"
      ),
      replicate = str_extract(filename, "\\d+$")
    ) %>%
    arrange(compound, blank_type, replicate)
}

#Compute mean and SD per compound × blank_type
compute_blank_stats <- function(df) {
  df %>%
    group_by(compound, blank_type) %>%
    summarise(
      blank_mean = mean(area, na.rm = TRUE),
      blank_sd   = sd(area, na.rm = TRUE),
      .groups = "drop"
    )
}

clean_post_spike <- function(df) {
  df %>%
    #Keep only "post" samples
    filter(str_detect(filename, "post")) %>%
    #Extract dose, tissue, and replicate
    mutate(
      dose = str_extract(filename, "10_nm|100_nm|1_um|10_um|100_um"),
      replicate = str_extract(filename, "\\d+$"),
      #Use area_norm if exists, otherwise area_corrected
      intensity = if ("area_norm" %in% names(.)) area_norm else area
    ) %>%
    #Pivot so all replicates for a compound/dose/tissue are on one row
    pivot_wider(
      id_cols = c(compound, dose),   #row identifier
      names_from = replicate,
      values_from = intensity,
      names_prefix = "rep"
    ) %>%
    #Order the doses
    mutate(
      dose = factor(dose, levels = c("10_nm", "100_nm", "1_um", "10_um", "100_um"))
    ) %>%
    arrange(compound, dose)
}

clean_samples <- function(df) {
  df %>%
    mutate(
      subj_id = as.integer(str_extract(filename, "\\d+")),
      #assign treatment based on subject ID
      treatment = case_when(
        subj_id %in% 1:5  ~ "4 hr",
        subj_id %in% 6:10 ~ "1 hr",
        subj_id %in% 11:15 ~ "2 hr",
        subj_id %in% 16:20 ~ "No Exposure",
        subj_id %in% 21:25 ~ "0 hr",
        subj_id %in% 26:31 ~ "24 hr",
        subj_id %in% 32:37 ~ "8 hr",
        TRUE ~ "No Exposure"   #default fallback
      ),
      replicate = row_number() %>% as.character()  #temporary replicate assignment
    ) %>%
    group_by(compound, treatment) %>%
    mutate(replicate = paste0("rep", row_number())) %>%  #true replicate numbering
    ungroup() %>%
    dplyr::select(compound, treatment, replicate, area_norm) %>%
    pivot_wider(
      names_from = replicate,
      values_from = area_norm,
      values_fn = list(area_norm = mean),   #or = list(area_norm = identity) if you want ALL values kept
      values_fill = NA
    ) %>%
    mutate(
      treatment = factor(treatment,
                         levels = c("No Exposure", "0 hr", "1 hr", "2 hr", "4 hr", "8 hr", "24 hr"))
    ) %>%
    arrange(compound, treatment)
}

        TRUE ~ "No Exposure"   #default fallback
      ),
      replicate = row_number() %>% as.character()  #temporary replicate assignment
    ) %>%
    group_by(compound, treatment) %>%
    mutate(replicate = paste0("rep", row_number())) %>%  #true replicate numbering
    ungroup() %>%
    dplyr::select(compound, treatment, replicate, area) %>%
    pivot_wider(
      names_from = replicate,
      values_from = area,
      values_fn = list(area = mean),   #or = list(area_norm = identity) if you want ALL values kept
      values_fill = NA
    ) %>%
    mutate(
      treatment = factor(treatment,
                         levels = c("No Exposure", "0 hr", "1 hr", "2 hr", "4 hr", "8 hr", "24 hr"))
